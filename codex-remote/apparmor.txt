# Isolated Codex SSH app profile.
#include <tunables/global>
profile codex_remote flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  network,
  capability net_bind_service,
  capability setuid,
  capability setgid,
  capability chown,
  capability dac_override,
  capability fowner,
  capability sys_chroot,
  /init rix,
  /package/** rix,
  /usr/** rix,
  /bin/** rix,
  /sbin/** rix,
  /lib/** mr,
  /lib64/** mr,
  /etc/** r,
  /run/** rw,
  /run/s6/** rix,
  /proc/** r,
  /dev/null rw,
  /dev/zero rw,
  /dev/urandom r,
  /dev/tty rw,
  /dev/ptmx rw,
  /dev/pts/ptmx rw,
  /dev/pts/[0-9]* rw,
  /data/** rwk,
  # Codex installs its managed app-server binary into its private data volume.
  /data/codex-home/packages/** ix,
  /tmp/** rwk,
  deny /config/** rwklx,
  deny /var/run/docker.sock rwklx,
  deny /run/docker.sock rwklx,
}
